' Abstract Factory
abstract class IPCFactory {
    +CreateBox(): Box*
    +CreateProcessor(): Processor*
    +CreateMainBoard(): MainBoard*
    +CreateHdd(): Hdd*
    +CreateMemory(): Memory*
}

' Concrete Factories
class HomePCFactory {
    +CreateBox(): Box*
    +CreateProcessor(): Processor*
    +CreateMainBoard(): MainBoard*
    +CreateHdd(): Hdd*
    +CreateMemory(): Memory*
}

class OfficePCFactory {
    +CreateBox(): Box*
    +CreateProcessor(): Processor*
    +CreateMainBoard(): MainBoard*
    +CreateHdd(): Hdd*
    +CreateMemory(): Memory*
}

' PC Class
class PC {
    -Box* box
    -Processor* processor
    -MainBoard* mainBoard
    -Hdd* hdd
    -Memory* memory
    +PC()
    +~PC()
    +setBox(Box*)
    +setProcessor(Processor*)
    +setMainBoard(MainBoard*)
    +setHdd(Hdd*)
    +setMemory(Memory*)
    +getBox(): Box*
    +getProcessor(): Processor*
    +getMainBoard(): MainBoard*
    +getHdd(): Hdd*
    +getMemory(): Memory*
    +print()
}

' Configurator Class
class PcConfigurator {
    -IPCFactory* pcFactory
    +PcConfigurator()
    +~PcConfigurator()
    +setPcFactory(IPCFactory*)
    +configure(): PC*
}
