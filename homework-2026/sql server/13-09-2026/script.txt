GO

USE CarSharingAnalytics;
GO

CREATE TABLE Trips (
    TripID INT PRIMARY KEY,
    StartDateTime DATETIME NOT NULL,
    EndDateTime DATETIME NOT NULL,
    DurationMinutes INT NOT NULL CHECK (DurationMinutes > 0),
    DistanceKm DECIMAL(6,2) NOT NULL,
    StartAddress NVARCHAR(150) NOT NULL,
    EndAddress NVARCHAR(150) NOT NULL,
    ClientName NVARCHAR(100) NOT NULL,
    ClientAge INT NOT NULL CHECK (ClientAge BETWEEN 18 AND 99),
    ClientGender NVARCHAR(10) NOT NULL,
    ClientCity NVARCHAR(50) NOT NULL,
    DrivingExperienceYears INT NOT NULL,
    CarBrand NVARCHAR(50) NOT NULL,
    CarModel NVARCHAR(50) NOT NULL,
    CarYear INT NOT NULL CHECK (CarYear BETWEEN 2000 AND YEAR(GETDATE())),
    CarColor NVARCHAR(30) NOT NULL,
    RentalPricePerMinute DECIMAL(6,2) NOT NULL CHECK (RentalPricePerMinute > 0),
    TotalCost DECIMAL(10,2) NOT NULL,
    HasDiscount BIT NOT NULL,
    DiscountPercent INT CHECK (DiscountPercent BETWEEN 0 AND 100),
    PaymentMethod NVARCHAR(20) NOT NULL,
    ClientRating INT CHECK (ClientRating BETWEEN 1 AND 5),
    HasComment BIT NOT NULL,
    HasViolation BIT NOT NULL,
    FineAmount DECIMAL(10,2) CHECK (FineAmount >= 0)
);
GO
